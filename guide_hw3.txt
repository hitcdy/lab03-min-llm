一、实验目的
1. 理解语言模型“预测下一个字符”的自监督训练目标，掌握字符级分词、上下文窗口
截取与“标签错开一位”（输入取前T 个、标签取后T 个）的实现；
2. 亲手用PyTorch 搭建一个约50 万参数的最小GPT（词嵌入+ 可学习位置编码+
Pre-Norm Transformer 块+ 权重共享），把教材第5 章的公式变成能跑的代码；
3. 在CPU 上完成基线训练，绘制loss 曲线，并用模型自回归生成“诗句”，直观感受
温度（temperature）与top-k 采样对生成质量的影响；
4. 通过控制变量实验（学习率、层数、嵌入维度、上下文长度、位置编码）理解各超
参数的作用，并与手算参数量相互印证；
5. 熟练使用TRAE IDE 的AI 对话、代码解释与错误修复能力，规范记录人机协作过
程。

六、实验步骤
步骤1
用TRAE Builder 生成实验框架：新建目录hw3_min_llm 与文件
min_llm.py；让AI 按第七节框架逐段生成代码。提示：模型内部所有张量形状为(B,
T, C)——批大小、序列长度、嵌入维度；注意力内部才拆出多头维。
步骤2
手算验证（必做）：用4.2 节公式计算基线总参数量，写出代入过程填入报
告，并用sum(p.numel() for p in model.parameters()) 的打印值核对。
步骤3
运行基线并保存结果：python min_llm.py（CPU 约20~30 分钟，本机实测约
1.3 it/s ≈25 分钟；时间紧可用--iters 800 先跑通）。保存loss 曲线与
generated_base.txt，把参数量、最终loss、耗时填入报告。
步骤4
逐组完成对照实验：按表5 逐组实验，每次只改一个变量；每组保存曲线
（脚本会自动按配置命名loss_curve_*.png，如--n_layer 1 对应
loss_curve_L1_E128_lr0.001.png），并把结果填入表5。
步骤5
采样实验与生成分析：固定基线模型，对比temperature = 0.5 / 1.0 / 1.5 与
top_k = 5 / 20 / 不限制（设为词表大小即可）共3~4 组生成，每组3 段；从“确定性↔
多样性”“重复↔胡言”两个维度分析，给出你认为最优的组合并说明理由。
步骤6（选做）
扩充语料并对比：按任务四(a) 追加自备语料，重跑基线，对比生
成内容与loss 曲线的变化。

# ============ 2. 字符级分词器============
class CharTokenizer:
def __init__(self, text):
chars = sorted(set(text))
self.stoi = {ch: i for i, ch in enumerate(chars)}
self.itos = {i: ch for ch, i in self.stoi.items()}
self.vocab_size = len(chars)
def encode(self, s):
return [self.stoi[c] for c in s if c in self.stoi]

def decode(self, ids):
return "".join(self.itos[i] for i in ids)
# ============ 3. 最小GPT 模型============
class CausalSelfAttention(nn.Module):
"""带因果掩码的多头自注意力（教材5.4 节）"""
def __init__(self, n_embd, n_head, block_size):
super().__init__()
assert n_embd % n_head == 0
self.n_head = n_head
self.qkv = nn.Linear(n_embd, 3 * n_embd)
# Q/K/V 一次算出
self.proj = nn.Linear(n_embd, n_embd)
# 输出投影W_O
self.register_buffer(
"mask",
torch.tril(torch.ones(block_size, block_size)).view(1, 1, block_size,
block_size))
def forward(self, x):
B, T, C = x.shape
q, k, v = self.qkv(x).split(C, dim=2)
# 拆多头: (B, T, C) -> (B, n_head, T, head_dim)
q = q.view(B, T, self.n_head, -1).transpose(1, 2)
k = k.view(B, T, self.n_head, -1).transpose(1, 2)
v = v.view(B, T, self.n_head, -1).transpose(1, 2)
att = (q @ k.transpose(-2, -1)) / math.sqrt(k.size(-1))
# 缩放点积
att = att.masked_fill(self.mask[:, :, :T, :T] == 0, float("-inf"))
# 因果掩码
att = F.softmax(att, dim=-1)
# 归一化
y = att @ v
# 加权求和
y = y.transpose(1, 2).contiguous().view(B, T, C)
# 拼接多头
return self.proj(y)
class Block(nn.Module):
"""Pre-Norm Transformer 块：LN -> 注意力-> 残差，LN -> FFN -> 残差"""
def __init__(self, n_embd, n_head, block_size):
super().__init__()
self.ln1 = nn.LayerNorm(n_embd)
self.attn = CausalSelfAttention(n_embd, n_head, block_size)
self.ln2 = nn.LayerNorm(n_embd)
self.mlp = nn.Sequential(
nn.Linear(n_embd, 4 * n_embd), nn.GELU(), nn.Linear(4 * n_embd, n_embd))
def forward(self, x):
x = x + self.attn(self.ln1(x))
# 残差连接
x = x + self.mlp(self.ln2(x))
return x
class MiniGPT(nn.Module):
def __init__(self, vocab_size, n_embd=128, n_head=4, n_layer=2,
block_size=128, use_pos=True):
super().__init__()
self.block_size = block_size
self.use_pos = use_pos
self.tok_emb = nn.Embedding(vocab_size, n_embd)
# 词嵌入
self.pos_emb = nn.Embedding(block_size, n_embd) if use_pos else None
# 可学习位置
编码
self.blocks = nn.ModuleList(
[Block(n_embd, n_head, block_size) for _ in range(n_layer)])
self.ln_f = nn.LayerNorm(n_embd)

self.head = nn.Linear(n_embd, vocab_size, bias=False)
self.head.weight = self.tok_emb.weight
# 权重共享
self.apply(self._init_weights)
# 小方差初始化
def _init_weights(self, m):
"""GPT 标准初始化：权重N(0, 0.02)，偏置清零→初始loss ≈ln V"""
if isinstance(m, nn.Linear):
nn.init.normal_(m.weight, mean=0.0, std=0.02)
if m.bias is not None:
nn.init.zeros_(m.bias)
elif isinstance(m, nn.Embedding):
nn.init.normal_(m.weight, mean=0.0, std=0.02)
def forward(self, idx, targets=None):
B, T = idx.shape
pos = torch.arange(T, device=idx.device)
x = self.tok_emb(idx)
if self.pos_emb is not None:
x = x + self.pos_emb(pos)
# 词向量+ 位置向量
for blk in self.blocks:
x = blk(x)
logits = self.head(self.ln_f(x))
loss = None
if targets is not None:
loss = F.cross_entropy(logits.view(-1, logits.size(-1)), targets.view(-1))
return logits, loss
@torch.no_grad()
def generate(self, idx, max_new_tokens=200, temperature=1.0, top_k=None):
"""自回归生成：每步预测下一个字符并拼接回输入（教材代码5-6 思想）"""
self.eval()
for _ in range(max_new_tokens):
idx_cond = idx[:, -self.block_size:]
# 截取上下文窗口
logits, _ = self(idx_cond)
logits = logits[:, -1, :] / max(temperature, 1e-8)
if top_k is not None:
kth = torch.topk(logits, min(top_k, logits.size(-1)))[0][:, -1:]
logits[logits < kth] = float("-inf")
# 只保留top-k
probs = F.softmax(logits, dim=-1)
next_id = torch.multinomial(probs, 1)
# 按概率采样
idx = torch.cat([idx, next_id], dim=1)
return idx
# ============ 4. 数据批生成：随机截取(block_size+1) 的片段============
def get_batch(data, block_size, batch_size, device):
ix = torch.randint(len(data) - block_size - 1, (batch_size,))
x = torch.stack([data[i:i + block_size] for i in ix])
y = torch.stack([data[i + 1:i + block_size + 1] for i in ix])
# 错开一位作标签
return x.to(device), y.to(device)

九、常见问题
Q1：CPU 训练一组要多久？
基线2000 步约20~30 分钟（本机实测1.3 it/s ≈25 分钟，视CPU 而定）；6 组对照用
1000 步每组约12~15 分钟，全套约2.5 小时。先--iters 800 跑通全流程，再跑全量；
多组实验可挂后台并行写报告。
Q2：loss 不降甚至NaN？
多为学习率过大（1e-2 可能发散）→回到1e-3；其次检查标签是否与输入错开一位、
交叉熵前是否把logits 展平成(-, V)。
Q3：为什么用字符级分词而不是BPE？
字符级词表小、零预处理、不需要额外分词库，能把注意力集中到架构本身；其缺点
（序列长、单token 信息少）正是思考题(4) 要讨论的内容。
Q4：训练时电脑很卡/ 风扇狂转？
PyTorch 默认占满所有CPU 核属正常现象。可在文件开头加torch.set_num_threads(4)
限制线程，把算力留给系统；训练时避免开大型软件。
Q5：生成内容一直循环重复？
温度过低（如0.5）或top_k 过小会加剧重复——这正是exp6 要观察的现象。进阶做
法见任务四(b)：重复惩罚。
Q6：小语料会把诗“背”下来？
会。语料仅几千字时，模型接近“记忆+ 插值”，loss 可降到很低、生成近乎原句——
这是小数据下的正常现象，也解释了真实LLM 为什么必须依靠海量数据才能“泛化”
而不是“背诵”（见思考题5）。
Q7：每次结果不一样？
固定--seed 42 即可复现。注意：若中途新增/ 修改了corpus_extra.txt，词表与数据都
会变化，对照实验各组必须保证语料一致。
